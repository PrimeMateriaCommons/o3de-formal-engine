definition{
  name: isActivationAdmissible,
  parameters: [reqId : Nat, incompId : Nat, currentActiveService : Nat],
  result: Bool,
  body: (((reqId =? 0) || (reqId =? currentActiveService)) && ((incompId =? 0) || (not (incompId =? currentActiveService))))
}

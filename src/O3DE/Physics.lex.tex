definition{
  name: integrateVelocityStep,
  parameters: [body : PhysXRigidBody, accelScaled : Nat, dtMs : Nat, dampingScaled : Nat],
  result: Nat,
  body: (if (project body isKinematic) then 0 else (let rawVel : Nat := ((project body velocityScaled) + (prim quotient (prim multiply accelScaled dtMs : Nat) 1000 0 : Nat)); (let dampedVel : Nat := (O3DE.Invariants::applyVelocityDamping rawVel dampingScaled); (if (dampedVel <=? (project body maxVelocityScaled)) then dampedVel else (project body maxVelocityScaled)))))
}

definition{
  name: computeKineticEnergyScaled,
  parameters: [massGrams : Nat, velocityScaled : Nat],
  result: Nat,
  body: (prim quotient (prim multiply massGrams (prim multiply velocityScaled velocityScaled : Nat) : Nat) 2000 0 : Nat)
}

theorem{
  name: kinematic_velocity_invariant_zero,
  parameters: [],
  statement: ((integrateVelocityStep (kinematicBody) 500 16 50) = 0),
  proof: reflexivity{}
}

  name: dynamic_velocity_step_calculation,
  parameters: [],

  proof: reflexivity{}
}

theorem{
  name: velocity_clamped_at_maximum,
  parameters: [],

  proof: reflexivity{}
}

theorem{
  name: kinetic_energy_soundness,
  parameters: [],
  statement: ((computeKineticEnergyScaled 2000 100) = 10000),
  proof: reflexivity{}
}

  name: isSubsteppingStable,
  parameters: [p : O3DEPhysicsParameters],
  result: Bool,
  body: (((4 <=? (project p positionSolverIterations)) && (1 <=? (project p velocitySolverIterations))) && (1 <=? (project p maxSubsteps)))
}
